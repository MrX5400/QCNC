            {sourceType === 'raster' && (
              <div className="space-y-4">
                {/* Karte A: Bildquelle & Vorbereitung */}
                <div className="p-3 bg-slate-900 rounded-xl border border-slate-700 shadow-sm space-y-3">
                  <div className="flex items-center gap-2 text-cyan-400 font-bold text-xs">
                    <ImageIcon className="w-4 h-4" /> Karte A: Bildquelle & Vorbereitung
                  </div>
                  <label className="border border-dashed border-slate-600 hover:border-cyan-500 rounded-lg p-3 flex flex-col items-center justify-center cursor-pointer transition-colors text-center bg-slate-950/50">
                    <span className="font-semibold text-slate-200 text-xs truncate max-w-[250px]">
                      {rasterImageName ? rasterImageName : 'Bild hochladen (PNG, JPG)'}
                    </span>
                    <input type="file" accept="image/*" onChange={handleRasterImageUpload} className="hidden" />
                  </label>
                  <div className="grid grid-cols-2 gap-2">
                    <button onClick={() => { setRasterSettings(s => ({...s, mode: 'centerline_trace', threshold: 160, detailSensitivity: 2, fillPattern: 'none'})); }} className="p-2 bg-slate-800 hover:bg-slate-700 rounded-lg text-[0.625rem] text-slate-300 font-semibold border border-slate-700 transition">🖋️ Handschrift / Text</button>
                    <button onClick={() => { setRasterSettings(s => ({...s, mode: 'contour_trace', threshold: 128, detailSensitivity: 5, fillPattern: 'none'})); }} className="p-2 bg-slate-800 hover:bg-slate-700 rounded-lg text-[0.625rem] text-slate-300 font-semibold border border-slate-700 transition">🎯 Logo / Silhouette</button>
                    <button onClick={() => { setRasterSettings(s => ({...s, mode: 'contour_trace', threshold: 200, detailSensitivity: 8, fillPattern: 'none'})); }} className="p-2 bg-slate-800 hover:bg-slate-700 rounded-lg text-[0.625rem] text-slate-300 font-semibold border border-slate-700 transition">📐 Technische Skizze</button>
                    <button onClick={() => { setRasterSettings(s => ({...s, mode: 'contour_trace', threshold: 128, detailSensitivity: 5, blurRadius: 2, fillPattern: 'crosshatch'})); }} className="p-2 bg-slate-800 hover:bg-slate-700 rounded-lg text-[0.625rem] text-slate-300 font-semibold border border-slate-700 transition">🖼️ Foto / Grafik</button>
                  </div>
                  <div className="flex items-center gap-2 pt-1 border-t border-slate-800">
                    <div className="flex-1 space-y-1">
                      <div className="flex justify-between text-[0.625rem] text-slate-400 font-semibold">
                        <span>Schwellenwert</span>
                        <span className="text-cyan-400">{rasterSettings.threshold}</span>
                      </div>
                      <input type="range" min="0" max="255" value={rasterSettings.threshold} onChange={e => setRasterSettings(s => ({...s, threshold: Number(e.target.value)}))} className="w-full accent-cyan-500 h-1.5" />
                    </div>
                    <button onClick={handleAutoOtsuThreshold} className="px-2 py-1.5 bg-cyan-600/20 text-cyan-400 rounded-lg text-[0.625rem] font-bold border border-cyan-500/30 hover:bg-cyan-600/40 transition mt-3">
                      ⚡ Auto
                    </button>
                  </div>
                </div>

                {/* Karte B: Vorschau & Ebenen-Überlagerung */}
                <div className="p-3 bg-slate-900 rounded-xl border border-slate-700 shadow-sm space-y-3">
                  <div className="flex items-center gap-2 text-indigo-400 font-bold text-xs">
                    <Eye className="w-4 h-4" /> Karte B: Live-Vorschau & Lupe
                  </div>
                  {rasterImage ? (
                    <>
                      <div 
                        className="h-36 rounded-lg bg-slate-950 border border-slate-800 relative cursor-pointer group flex items-center justify-center overflow-hidden"
                        onClick={() => setShowImageLightbox(true)}
                        title="Klicken für interaktive Vollbild-Vorschau"
                      >
                        <canvas ref={vectorOverlayCanvasRef} className="max-w-full max-h-full object-contain z-10" />
                        <img src={rasterImage.src} className="absolute max-w-full max-h-full object-contain opacity-20 pointer-events-none" alt="" />
                        <div className="absolute inset-0 bg-black/50 opacity-0 group-hover:opacity-100 flex items-center justify-center transition-opacity z-20 backdrop-blur-sm">
                          <div className="flex items-center gap-2 text-white bg-indigo-600/80 px-4 py-2 rounded-full font-bold shadow-lg">
                            <Expand className="w-4 h-4" /> Großansicht öffnen
                          </div>
                        </div>
                      </div>
                    </>
                  ) : (
                    <div className="h-24 bg-slate-950 rounded-lg border border-slate-800 flex items-center justify-center text-slate-500 text-xs font-semibold">Kein Bild geladen</div>
                  )}
                </div>

                {/* Karte C: Vektorisierungs-Modus & Geometrie */}
                <div className="p-3 bg-slate-900 rounded-xl border border-slate-700 shadow-sm space-y-3">
                  <div className="flex items-center gap-2 text-emerald-400 font-bold text-xs">
                    <Workflow className="w-4 h-4" /> Karte C: Vektor-Modus & Maße
                  </div>
                  
                  <div className="grid grid-cols-2 gap-2">
                    <button 
                      onClick={() => setRasterSettings(s => ({...s, mode: 'contour_trace'}))}
                      className={`p-2 rounded-lg border text-xs font-bold flex flex-col items-center justify-center gap-1 transition ${rasterSettings.mode === 'contour_trace' ? 'bg-emerald-950 border-emerald-500 text-emerald-300 shadow-sm' : 'bg-slate-800 border-slate-700 text-slate-400 hover:bg-slate-700'}`}
                    >
                      <Search className="w-4 h-4" /> Kontur (Potrace)
                    </button>
                    <button 
                      onClick={() => setRasterSettings(s => ({...s, mode: 'centerline_trace'}))}
                      className={`p-2 rounded-lg border text-xs font-bold flex flex-col items-center justify-center gap-1 transition ${rasterSettings.mode === 'centerline_trace' ? 'bg-emerald-950 border-emerald-500 text-emerald-300 shadow-sm' : 'bg-slate-800 border-slate-700 text-slate-400 hover:bg-slate-700'}`}
                    >
                      <Move className="w-4 h-4" /> Mittellinie (1-Strich)
                    </button>
                  </div>

                  <div className="space-y-1">
                    <div className="flex justify-between text-[0.625rem] text-slate-400 font-semibold">
                      <span>Kurvenglättung</span>
                      <span className="text-emerald-400">{rasterSettings.detailSensitivity}</span>
                    </div>
                    <input type="range" min="0" max="10" value={rasterSettings.detailSensitivity || 5} onChange={e => setRasterSettings(s => ({...s, detailSensitivity: Number(e.target.value)}))} className="w-full accent-emerald-500 h-1.5" />
                  </div>

                  <div className="grid grid-cols-2 gap-3 pt-2 border-t border-slate-700/50">
                    <div className="space-y-1">
                      <span className="text-[0.625rem] text-slate-400 font-semibold">Breite (mm)</span>
                      <input type="number" min={1} value={rasterSettings.targetWidth} onChange={e => handleRasterWidthChange(Number(e.target.value))} className="w-full bg-slate-950 text-white rounded p-1.5 text-xs font-bold border border-slate-700 focus:border-emerald-500 outline-none transition" />
                    </div>
                    <div className="space-y-1">
                      <div className="flex items-center justify-between">
                        <span className="text-[0.625rem] text-slate-400 font-semibold">Höhe (mm)</span>
                        <button onClick={() => setRasterLockAspect(!rasterLockAspect)} className="text-slate-400 hover:text-emerald-400 transition" title="Proportionen sperren">
                          {rasterLockAspect ? <Lock className="w-3 h-3 text-emerald-400" /> : <Unlock className="w-3 h-3" />}
                        </button>
                      </div>
                      <input type="number" min={1} value={rasterSettings.targetHeight} onChange={e => handleRasterHeightChange(Number(e.target.value))} className="w-full bg-slate-950 text-white rounded p-1.5 text-xs font-bold border border-slate-700 focus:border-emerald-500 outline-none transition" />
                    </div>
                  </div>
                </div>

                {/* Karte D: Füllmuster */}
                <div className="p-3 bg-slate-900 rounded-xl border border-slate-700 shadow-sm space-y-3">
                  <div className="flex items-center gap-2 text-purple-400 font-bold text-xs">
                    <GridIcon className="w-4 h-4" /> Karte D: Füllmuster (Hatching)
                  </div>
                  
                  <select 
                    value={rasterSettings.fillPattern || 'none'} 
                    onChange={e => setRasterSettings(s => ({...s, fillPattern: e.target.value as any}))}
                    className="w-full bg-slate-950 text-slate-200 rounded-lg p-2 text-xs border border-slate-700 font-semibold focus:border-purple-500 outline-none transition"
                  >
                    <option value="none">🚫 Keine / Nur Kontur</option>
                    <option value="lines">📏 Parallele Schraffur</option>
                    <option value="crosshatch">✖️ Kreuzgitter</option>
                    <option value="zigzag">⚡ Zick-Zack</option>
                    <option value="concentric">🎯 Konzentrisch</option>
                    <option value="spiral">🌀 Spirale (Archimedes)</option>
                    <option value="stippling">░ Punktraster (Dither)</option>
                    <option value="wave">〰️ Wellenlinien</option>
                    <option value="honeycomb">⬡ Wabenmuster</option>
                  </select>

                  {rasterSettings.fillPattern && rasterSettings.fillPattern !== 'none' && (
                    <div className="space-y-3 pt-2">
                      <div className="space-y-1">
                        <div className="flex justify-between text-[0.625rem] text-slate-400 font-semibold">
                          <span>Linienabstand (mm)</span>
                          <span className="text-purple-400">{rasterSettings.fillSpacing || 2}</span>
                        </div>
                        <input type="range" min="0.2" max="10" step="0.2" value={rasterSettings.fillSpacing || 2} onChange={e => setRasterSettings(s => ({...s, fillSpacing: Number(e.target.value)}))} className="w-full accent-purple-500 h-1.5" />
                      </div>
                      
                      <div className="space-y-1">
                        <div className="flex justify-between text-[0.625rem] text-slate-400 font-semibold">
                          <span>Drehwinkel (°)</span>
                          <span className="text-purple-400">{rasterSettings.fillAngle || 45}°</span>
                        </div>
                        <input type="range" min="0" max="180" step="5" value={rasterSettings.fillAngle || 45} onChange={e => setRasterSettings(s => ({...s, fillAngle: Number(e.target.value)}))} className="w-full accent-purple-500 h-1.5" />
                      </div>
                      
                      <label className="flex items-center gap-2 text-xs font-semibold text-slate-300 cursor-pointer pt-1 border-t border-slate-800">
                        <input type="checkbox" checked={rasterSettings.fillIncludeContour ?? true} onChange={e => setRasterSettings(s => ({...s, fillIncludeContour: e.target.checked}))} className="rounded bg-slate-900 border-slate-700 text-purple-500 focus:ring-purple-500 w-4 h-4 cursor-pointer transition" />
                        Außenkontur mitzeichnen
                      </label>
                    </div>
                  )}
                </div>

                <button 
                  onClick={handleAddCurrentToComposition}
                  disabled={rawPolylines.length === 0}
                  className="w-full py-3 bg-cyan-600 hover:bg-cyan-500 disabled:bg-slate-800 disabled:text-slate-600 text-white rounded-xl font-bold flex justify-center items-center gap-2 shadow-lg transition-all mt-4"
                >
                  <Plus className="w-5 h-5" /> Vektoren zur Arbeitsfläche hinzufügen
                </button>
              </div>
            )}
